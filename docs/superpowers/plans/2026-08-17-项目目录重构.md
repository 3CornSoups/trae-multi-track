```python
# -*- coding: utf-8 -*-
"""压缩归档：paper-project / citation_network / 匹配_extracted(排除匹配/子目录)"""
import zipfile, os
from pathlib import Path

ROOT = Path(r"D:\论文和代码项目\论文\TRAE\多轨道")
ARCH = ROOT / "_archived"
ARCH.mkdir(exist_ok=True)

def zip_dir(src: Path, out: Path, exclude_subdirs: tuple = ()):
    with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED, compresslevel=6) as z:
        for f in sorted(src.rglob("*")):
            rel = f.relative_to(src)
            if any(rel.parts[0] == e for e in exclude_subdirs):
                continue
            if f.is_file():
                z.write(f, arcname=f"{src.name}/{rel.as_posix()}")

zip_dir(ROOT / "paper-project", ARCH / "paper-project.zip")
zip_dir(ROOT / "citation_network", ARCH / "citation_network.zip")
zip_dir(ROOT / "匹配_extracted", ARCH / "匹配_extracted.zip", exclude_subdirs=("匹配",))
print("DONE")
```
